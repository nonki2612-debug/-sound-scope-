                <div>・URLプロトコル: <span class="${protocol === 'https:' ? 'text-emerald-400' : 'text-rose-400'}">${protocol}</span></div>

                ${errorMessage ? `<div class="text-rose-400 mt-1">・エラー詳細: ${errorMessage}</div>` : ''}

        }

        async function initMicrophone() {
            try {
                // セキュア環境チェック
                if (window.location.protocol === 'file:') {
                    throw new Error('iPadの「ファイル」から直接開かれています。file://環境ではiPadのマイク権限は動作しません。');
                }

                if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                    throw new Error('お使いのブラウザ環境ではマイク機能へのアクセスが制限されています（HTTPSが必要です）。');
                }

                const AudioContext = window.AudioContext || window.webkitAudioContext;
                state.audioContext = new AudioContext();

                // iOS Safari対策: ユーザー操作直後に必ずresumeを実行
                if (state.audioContext.state === 'suspended') {
                    await state.audioContext.resume();
                }

                // iOS互換性向上のため、最初は最も安全な { audio: true } で要求
                let stream;
                try {
                    stream = await navigator.mediaDevices.getUserMedia({ audio: true });
                } catch (e) {
                    // フォールバック
                    stream = await navigator.mediaDevices.getUserMedia({
                        audio: {
                            echoCancellation: false,
                            noiseSuppression: false
                        }
                    });
                }

                state.microphoneStream = stream;
                const source = state.audioContext.createMediaStreamSource(stream);

                state.analyser = state.audioContext.createAnalyser();
                state.analyser.fftSize = 2048;
                source.connect(state.analyser);

                state.timeData = new Float32Array(state.analyser.fftSize);

                // 再生用に約1.5秒分のリングバッファを準備
                const sampleRate = state.audioContext.sampleRate;
                state.ringBufferSize = Math.floor(sampleRate * 1.5);
                state.ringBuffer = new Float32Array(state.ringBufferSize);
                state.ringBufferIndex = 0;

                // スピーカーへのハウリング（ループバック）を完全に遮断するためのゲイン0ノード
                state.silentGain = state.audioContext.createGain();
                state.silentGain.gain.setValueAtTime(0, state.audioContext.currentTime);

                state.scriptProcessor = state.audioContext.createScriptProcessor(2048, 1, 1);
                state.scriptProcessor.onaudioprocess = (e) => {
                    const input = e.inputBuffer.getChannelData(0);
                    for (let i = 0; i < input.length; i++) {
                        state.ringBuffer[state.ringBufferIndex] = input[i];
                        state.ringBufferIndex = (state.ringBufferIndex + 1) % state.ringBufferSize;
                    }
                };
                source.connect(state.scriptProcessor);
                // WebKitではdestinationに接続しないとProcessorが停止するため、無音ゲイン経由で接続
                state.scriptProcessor.connect(state.silentGain);
                state.silentGain.connect(state.audioContext.destination);

                state.isMicRunning = true;
                state.isDemoMode = false;

                // UI更新
                startOverlay.classList.add('hidden');
                captureBtn.disabled = false;
                micStatusBadge.className = 'inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-xs font-medium bg-emerald-100 text-emerald-700';
                micStatusBadge.innerHTML = '<span class="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></span> マイク入力中';

                showToast('マイクが接続されました！音を出してみてください。', 'success');
                requestAnimationFrame(drawScope);
            } catch (err) {
                console.error("マイク取得エラー:", err);
                const errorDesc = err.name ? `${err.name}: ${err.message}` : err.toString();
                updateDiagnostics(errorDesc);

                micStatusBadge.className = 'inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-xs font-medium bg-rose-100 text-rose-700';
                micStatusBadge.innerHTML = '<span class="w-2 h-2 rounded-full bg-rose-500"></span> マイク接続エラー';

                showToast('マイクに接続できませんでした。診断パネルをご確認ください。', 'warn');
                micHelpModal.classList.remove('hidden');
            }
        }

        // デモ用波形ジェネレータ（数学的にリアルな波形を生成）
        function generateDemoSample(t, type) {
            // t: 秒数
            if (type === 'fork') {
                // 440Hz 単純正弦波
                return 0.7 * Math.sin(2 * Math.PI * 440 * t);
            } else if (type === 'high') {
                // 880Hz（高い声）+ 少しの倍音
                return 0.6 * Math.sin(2 * Math.PI * 880 * t) + 0.2 * Math.sin(2 * Math.PI * 1760 * t);
            } else if (type === 'low') {
                // 220Hz（低い声）+ 豊かな倍音
                return 0.7 * Math.sin(2 * Math.PI * 220 * t) + 0.3 * Math.sin(2 * Math.PI * 440 * t) + 0.15 * Math.sin(2 * Math.PI * 660 * t);
            } else if (type === 'recorder') {
                // 523.25Hz (C5) + 奇数倍音（管楽器風の波形）
                return 0.5 * Math.sin(2 * Math.PI * 523.25 * t) +
                       0.3 * Math.sin(2 * Math.PI * 1046.5 * t) +
                       0.2 * Math.sin(2 * Math.PI * 1569.75 * t);
            }
            return 0;
        }

        /**
         * ゼロクロス検出（トリガー機能）
         * 波の位相（山や谷）が左端で揃うように開始インデックスを探す。
         * これにより中学生が画面を見比べたとき、一目で波の形の違いがわかります。
         */
        function findZeroCrossing(data, bufferLen) {
            let startIndex = 0;
            // 探索範囲は前半部分
            const searchLimit = Math.min(bufferLen / 2, 512);

            for (let i = 0; i < searchLimit; i++) {
                // 負から正へのクロス（ゼロ点）を探す
                if (data[i] < 0 && data[i + 1] >= 0) {
                    startIndex = i;
                    break;
                }
            }
            return startIndex;
        }

        // 波形キャプチャ機能
        function captureWaveform() {
            if (!state.isMicRunning) return;

            // 最新の波形データを取得
            if (!state.isDemoMode && state.analyser) {
                state.analyser.getFloatTimeDomainData(state.timeData);
            }

            // 空いているスロットを探す（なければ最古のスロット0を上書き）
            let targetSlot = state.slots.findIndex(s => s === null);
            if (targetSlot === -1) {
                targetSlot = 0;
            }

            // トリガー処理を行って波形の先頭を揃えた配列を作成
            const startIndex = findZeroCrossing(state.timeData, state.timeData.length);
            const capturedLength = 1024;
            const capturedArray = new Float32Array(capturedLength);

            for (let i = 0; i < capturedLength; i++) {
                const srcIdx = startIndex + i;
                capturedArray[i] = srcIdx < state.timeData.length ? state.timeData[srcIdx] : 0;
            }

            // 直近約1.2秒の音声を AudioBuffer に変換して保存
            let audioBuffer = null;
            if (state.audioContext && state.ringBuffer) {
                const sampleRate = state.audioContext.sampleRate || 44100;
                const recSamples = Math.min(state.ringBufferSize, Math.floor(sampleRate * 1.2));
                audioBuffer = state.audioContext.createBuffer(1, recSamples, sampleRate);
                const channelData = audioBuffer.getChannelData(0);

                if (state.isDemoMode) {
                    // デモモード用の音声を生成して保存
                    for (let i = 0; i < recSamples; i++) {
                        const t = i / sampleRate;
                        channelData[i] = generateDemoSample(t, state.demoSoundType);
                    }
                } else {
                    // リングバッファから時系列順に展開
                    const startPos = (state.ringBufferIndex - recSamples + state.ringBufferSize) % state.ringBufferSize;
                    for (let i = 0; i < recSamples; i++) {
                        const readIdx = (startPos + i) % state.ringBufferSize;
                        channelData[i] = state.ringBuffer[readIdx];
                    }
                }
            }

            // ラベルのデフォルト調整（デモモード時は音源名を付与）
            if (state.isDemoMode) {
                const soundNames = {
                    fork: 'おんさ (440Hz)',
                    high: '高い声 (880Hz)',
                    low: '低い声 (220Hz)',
                    recorder: 'リコーダー'
                };
                state.slotLabels[targetSlot] = soundNames[state.demoSoundType] || `データ ${targetSlot + 1}`;
            }

            // 記録データ作成
            state.slots[targetSlot] = {
                data: capturedArray,
                audioBuffer: audioBuffer,
                timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' })
            };
            state.slotVisible[targetSlot] = true;

            renderSlotsUI();
            showToast(`${SLOT_COLORS[targetSlot].name} に波形と音声を記録しました`, 'info');
        }

        // スロット音声の再生・停止処理
        function playSlotAudio(index) {
            const slot = state.slots[index];
            if (!slot || !slot.audioBuffer || !state.audioContext) return;

            // すでに再生中の音があれば停止
            if (state.currentPlayingSource) {
                try {
                    state.currentPlayingSource.stop();
                } catch(e) {}
                state.currentPlayingSource = null;
            }

            // 同じスロットを押した場合はトグル停止
            if (state.playingSlotIndex === index) {
                state.playingSlotIndex = -1;
                renderSlotsUI();
                return;
            }

            if (state.audioContext.state === 'suspended') {
                state.audioContext.resume();
            }

            const source = state.audioContext.createBufferSource();
            source.buffer = slot.audioBuffer;

            // 音量調整用のゲイン
            const playGain = state.audioContext.createGain();
            playGain.gain.setValueAtTime(1.2, state.audioContext.currentTime);

            source.connect(playGain);
            playGain.connect(state.audioContext.destination);

            state.currentPlayingSource = source;
            state.playingSlotIndex = index;
            renderSlotsUI();

            source.onended = () => {
                if (state.playingSlotIndex === index) {
                    state.playingSlotIndex = -1;
                    state.currentPlayingSource = null;
                    renderSlotsUI();
                }
            };

            source.start(0);
        }

        function drawGrid(w, h) {
            ctx.save();
            ctx.strokeStyle = '#1e293b'; // 暗いグリッド線
            ctx.lineWidth = 1;

            // 縦線
            const vDivisions = 10;
            for (let i = 0; i <= vDivisions; i++) {
                const x = (w / vDivisions) * i;
                ctx.beginPath();
                ctx.moveTo(x, 0);
                ctx.lineTo(x, h);
                ctx.stroke();
            }

            // 横線
            const hDivisions = 8;
            for (let i = 0; i <= hDivisions; i++) {
                const y = (h / hDivisions) * i;
                ctx.beginPath();
                ctx.moveTo(0, y);
                ctx.lineTo(w, y);
                ctx.stroke();
            }

            // センターライン（基準軸・中心線）
            ctx.strokeStyle = '#334155';
            ctx.lineWidth = 1.5;
            ctx.setLineDash([4, 4]);
            ctx.beginPath();
            ctx.moveTo(0, h / 2);
            ctx.lineTo(w, h / 2);
            ctx.stroke();
            ctx.restore();
        }

        // 波形描画メインループ
        function drawScope() {
            if (!state.isMicRunning) return;

            const rect = canvas.getBoundingClientRect();
            const w = rect.width;
            const h = rect.height;

            // 背景クリア
            ctx.fillStyle = '#090d16';
            ctx.fillRect(0, 0, w, h);

            // 方眼目盛りの描画
            if (state.showGrid) {
                drawGrid(w, h);
            }

            // デモモード時の模擬波形生成
            if (state.isDemoMode) {
                const sampleRate = 44100;
                state.demoTime = (state.demoTime || 0) + (1024 / sampleRate);
                for (let i = 0; i < state.timeData.length; i++) {
                    const t = state.demoTime + (i / sampleRate);
                    // わずかなランダムノイズを足してリアルな波形に近づける
                    const noise = (Math.random() - 0.5) * 0.03;
                    state.timeData[i] = generateDemoSample(t, state.demoSoundType) + noise;
                }
            } else if (state.analyser) {
                // 現在のマイクリアルタイム波形を取得
                state.analyser.getFloatTimeDomainData(state.timeData);
            }

            const liveStart = findZeroCrossing(state.timeData, state.timeData.length);

            // 1. リアルタイム波形の描画（半透明の白/シアンで背景表示）
            ctx.save();
            ctx.lineWidth = 2;
            ctx.strokeStyle = state.isDemoMode ? 'rgba(251, 191, 36, 0.7)' : 'rgba(255, 255, 255, 0.4)';
            ctx.beginPath();

            // 表示するサンプル数（ズーム倍率で調整）
            const displaySamples = Math.floor(600 / state.zoom);
            const step = w / displaySamples;

            for (let i = 0; i < displaySamples; i++) {
                const dataIndex = liveStart + i;
                const sample = dataIndex < state.timeData.length ? state.timeData[dataIndex] : 0;
                // 振幅ゲインを適用
                const y = (h / 2) - (sample * (h / 2) * state.gain);
                const x = i * step;

                if (i === 0) ctx.moveTo(x, y);
                else ctx.lineTo(x, y);
            }
            ctx.stroke();
            ctx.restore();

            // 2. 記録された最大5つの波形を重ねて描画
            state.slots.forEach((slot, index) => {
                if (!slot || !state.slotVisible[index]) return;

                const colorInfo = SLOT_COLORS[index];
                const slotData = slot.data;

                ctx.save();
                ctx.lineWidth = 2.5;
                ctx.strokeStyle = colorInfo.hex;
                ctx.shadowColor = colorInfo.hex;
                ctx.shadowBlur = 6;
                ctx.beginPath();

                for (let i = 0; i < displaySamples; i++) {
                    const sample = i < slotData.length ? slotData[i] : 0;
                    const y = (h / 2) - (sample * (h / 2) * state.gain);
                    const x = i * step;

                    if (i === 0) ctx.moveTo(x, y);
                    else ctx.lineTo(x, y);
                }
                ctx.stroke();
                ctx.restore();
            });

            // リアルタイム表示ラベル
            ctx.save();
            ctx.font = '11px sans-serif';
            ctx.fillStyle = state.isDemoMode ? '#f59e0b' : 'rgba(255, 255, 255, 0.5)';
            ctx.fillText(state.isDemoMode ? '★ 模擬実験波形を出力中' : '白: リアルタイムマイク入力', 12, 20);
            ctx.restore();

            requestAnimationFrame(drawScope);
        }

        // お手本音叉（440Hz）の生成機能
        function toggleReferenceTone() {
            if (!state.audioContext) {
                const AudioContext = window.AudioContext || window.webkitAudioContext;
                state.audioContext = new AudioContext();
            }

            if (state.isTonePlaying) {
                // 停止
                if (state.toneOscillator) {
                    state.toneOscillator.stop();
                    state.toneOscillator.disconnect();
                    state.toneOscillator = null;
                }
                state.isTonePlaying = false;
                toneBtn.classList.remove('bg-amber-600');
                toneBtn.classList.add('bg-indigo-800');
                toneBtn.innerHTML = '<i class="fa-solid fa-bell"></i> <span>おんさの音 (440Hz)</span>';
            } else {
                // 再生
                if (state.audioContext.state === 'suspended') {
                    state.audioContext.resume();
                }
                const osc = state.audioContext.createOscillator();
                const gainNode = state.audioContext.createGain();

                osc.type = 'sine'; // きれいな正弦波（おんさの音）
                osc.frequency.setValueAtTime(440, state.audioContext.currentTime); // ラ(A4) = 440Hz
                gainNode.gain.setValueAtTime(0.15, state.audioContext.currentTime); // 音量は控えめ

                osc.connect(gainNode);
                gainNode.connect(state.audioContext.destination);

                // マイク入力に流し込む（analyserにもつなげる）
                if (state.analyser) {
                    gainNode.connect(state.analyser);
                }

                osc.start();
                state.toneOscillator = osc;
                state.isTonePlaying = true;

                toneBtn.classList.remove('bg-indigo-800');
                toneBtn.classList.add('bg-amber-600');
                toneBtn.innerHTML = '<i class="fa-solid fa-bell-slash"></i> <span>おんさを止める</span>';
                showToast('440Hzの純音を出力中（きれいな正弦波）', 'info');
            }
        }

        // イベント登録
        startMicBtn.addEventListener('click', initMicrophone);
        startDemoModeBtn.addEventListener('click', startDemoMode);
        captureBtn.addEventListener('click', captureWaveform);

        // デモ音源切り替えボタン
        document.querySelectorAll('.demo-sound-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                const targetBtn = e.target.closest('button');
                const type = targetBtn.dataset.demoSound;
                state.demoSoundType = type;

                document.querySelectorAll('.demo-sound-btn').forEach(b => {
                    b.classList.remove('bg-amber-100', 'border-amber-400');
                    b.classList.add('bg-slate-50', 'border-slate-200');
                });
                targetBtn.classList.remove('bg-slate-50', 'border-slate-200');
                targetBtn.classList.add('bg-amber-100', 'border-amber-400');

                showToast(`音源を切り替えました: ${targetBtn.textContent.trim()}`, 'info');
            });
        });

        clearAllBtn.addEventListener('click', () => {
            if (state.currentPlayingSource) {
                try { state.currentPlayingSource.stop(); } catch(e){}
                state.playingSlotIndex = -1;
            }
            state.slots = [null, null, null, null, null];
            renderSlotsUI();
            showToast('すべての記録波形を消去しました', 'info');
        });

        zoomRange.addEventListener('input', (e) => {
            state.zoom = parseFloat(e.target.value);
            zoomVal.textContent = state.zoom.toFixed(1) + 'x';
        });

        gainRange.addEventListener('input', (e) => {
            state.gain = parseFloat(e.target.value);
            gainVal.textContent = state.gain.toFixed(1) + 'x';
        });

        gridToggle.addEventListener('change', (e) => {
            state.showGrid = e.target.checked;
        });

        guideBtn.addEventListener('click', () => guideModal.classList.remove('hidden'));
        closeGuideBtn?.addEventListener('click', () => guideModal.classList.add('hidden'));
        closeGuideBtn2?.addEventListener('click', () => guideModal.classList.add('hidden'));
        toneBtn.addEventListener('click', toggleReferenceTone);

        micHelpHeaderBtn.addEventListener('click', () => micHelpModal.classList.remove('hidden'));
        micHelpOverlayBtn.addEventListener('click', () => micHelpModal.classList.remove('hidden'));
        closeMicHelpBtn?.addEventListener('click', () => micHelpModal.classList.add('hidden'));
        closeMicHelpBtn2?.addEventListener('click', () => micHelpModal.classList.add('hidden'));

        // キーボードショートカット（スペースキーでキャプチャ可能）
        window.addEventListener('keydown', (e) => {
            if (e.code === 'Space' && !e.repeat && state.isMicRunning) {
                // テキストボックス入力中は無効化
                if (['INPUT', 'TEXTAREA'].includes(document.activeElement.tagName)) return;
                e.preventDefault();
                captureWaveform();
            }
        });

            resizeCanvas();
            renderSlotsUI();
            updateDiagnostics();
